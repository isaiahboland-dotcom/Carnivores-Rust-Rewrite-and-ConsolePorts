cmake_minimum_required(VERSION 3.13)
project(c2_wiiu_perf_probe C ASM)
set(RUST_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/target/wiiu-terrain.a")
if(NOT EXISTS "${RUST_ARCHIVE}")
  message(FATAL_ERROR "Build the Rust archive first; see README.md")
endif()
configure_file(terrain.gsh terrain.gsh COPYONLY)
configure_file(terrain_opaque.gsh terrain_opaque.gsh COPYONLY)
configure_file(menu.gsh menu.gsh COPYONLY)
set_source_files_properties(shader.S PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/terrain.gsh;${CMAKE_CURRENT_BINARY_DIR}/terrain_opaque.gsh;${CMAKE_CURRENT_BINARY_DIR}/menu.gsh")
add_executable(c2-wiiu-perf-probe main.c shader.S)
option(C2_RANGE_UPLOAD "Flush only written dynamic geometry ranges" ON)
target_compile_definitions(c2-wiiu-perf-probe PRIVATE C2_RANGE_UPLOAD=$<BOOL:${C2_RANGE_UPLOAD}>)
target_link_libraries(c2-wiiu-perf-probe PRIVATE "${RUST_ARCHIVE}" m)
wut_create_rpx(c2-wiiu-perf-probe)
