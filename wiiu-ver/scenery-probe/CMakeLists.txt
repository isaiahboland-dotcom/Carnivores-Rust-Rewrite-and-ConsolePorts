cmake_minimum_required(VERSION 3.13)
project(c2_wiiu_scenery_probe C ASM)
set(RUST_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/target/wiiu-terrain.a")
if(NOT EXISTS "${RUST_ARCHIVE}")
  message(FATAL_ERROR "Build the Rust archive first; see README.md")
endif()
configure_file(terrain.gsh terrain.gsh COPYONLY)
add_executable(c2-wiiu-scenery-probe main.c shader.S)
target_link_libraries(c2-wiiu-scenery-probe PRIVATE "${RUST_ARCHIVE}")
wut_create_rpx(c2-wiiu-scenery-probe)
